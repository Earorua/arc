import assert from "node:assert/strict";
import { execFileSync } from "node:child_process";
import { createHash } from "node:crypto";
import { readFileSync, writeFileSync } from "node:fs";

const work = "work/grounded-verification/";
const prefix = "experience-priority-quality-20261109-";
const phase = process.argv[2];
assert.ok(["before", "after"].includes(phase));
const sha = bytes => createHash("sha256").update(bytes).digest("hex");
const hash = path => sha(readFileSync(path));
const git = args => execFileSync("git", args, { encoding: "utf8", windowsHide: true }).trim();
const frozenPath = work + "experience-priority-author-20261109-resume-frozen-source.json";

const binding = JSON.parse(readFileSync(work + frozen.sourceBinding.path));
assert.equal(git(["rev-parse", "HEAD"]), "714ea9e54d87b91a1fee4883176d4f1dd06ecf63");
assert.equal(git(["branch", "--show-current"]), "codex/materials-replan-recovery-20261106");
assert.equal(git(["status", "--short", "--", "app", "tests"]), "");
const verify = (hashes, directory = "") => {
  for (const [path, expected] of Object.entries(hashes)) assert.equal(hash(directory + path), expected, path);
};
verify(frozen.sourceHashes);
assert.equal(Object.keys(frozen.sourceHashes).length, 9);
assert.deepEqual(frozen.sourceHashes, binding.sourceHashes);
verify(frozen.unchangedCandidateAndSharedHashes);

verify(frozen.harnessInputs, work);
verify(frozen.preservedPreviousArtifacts, work);
assert.equal(Object.keys(frozen.preservedPreviousArtifacts).length, 370);
for (const [path, expected] of Object.entries(frozen.sourceHashes)) {

}

assert.ok(readFileSync("tests/lib/materials/experience-evidence.test.ts").subarray(0, originalTests.length).equals(originalTests));
