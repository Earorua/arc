import assert from "node:assert/strict";
import { createHash } from "node:crypto";
import { execFileSync } from "node:child_process";
import { readFileSync, readdirSync, writeFileSync, mkdtempSync } from "node:fs";
import { resolve } from "node:path";
import { build } from "esbuild";
import { Miniflare, Log, LogLevel } from "miniflare";

const output = process.argv[2];
assert.match(output ?? "", /^work\/grounded-verification\/experience-priority-history-create-[a-z0-9-]+-results\.json$/u);
const sha = value => createHash("sha256").update(value).digest("hex");
const workerPath = "work/grounded-verification/experience-priority-native-worker-20261108.ts";
const compiled = await build({ entryPoints: [workerPath], bundle: true, write: false, format: "esm", platform: "browser",
  target: "es2022", external: ["cloudflare:workers", "node:*"], metafile: true });
const script = compiled.outputFiles[0].text;
const persist = mkdtempSync(resolve("work/grounded-verification/experience-priority-history-store-20261108-"));
let outboundCalls = 0;
const runtime = new Miniflare({ cf: false, modules: true, script, compatibilityDate: "2026-05-15", compatibilityFlags: ["nodejs_compat"],
  d1Databases: { DB: "experience-priority-history" }, d1Persist: persist + "/d1", r2Buckets: { DOCS: "experience-priority-history-docs" }, r2Persist: persist + "/r2",
  log: new Log(LogLevel.ERROR), outboundService: async () => { outboundCalls++; throw Error("Outbound denied"); } });
try {
  const db = await runtime.getD1Database("DB");
  for (const file of readdirSync("drizzle").filter(name => /^\d{4}_.+\.sql$/u.test(name) && Number(name.slice(0,4)) <= 15).sort()) {
    for (const sql of readFileSync("drizzle/" + file, "utf8").split("--> statement-breakpoint").map(value => value.trim()).filter(Boolean)) await db.prepare(sql).run();
  }
  await db.prepare("INSERT INTO users(id,name,email,email_verified) VALUES('owner-a','Synthetic owner','owner-a@example.test',0)").run();
  const response = await runtime.dispatchFetch("http://localhost/history", { method: "POST", body: JSON.stringify({ action: "create-history" }) });
  const observed = await response.json();
  const rows = (await db.prepare("SELECT * FROM material_context_versions ORDER BY id").all()).results;
  const inputs = Object.fromEntries(Object.keys(compiled.metafile.inputs).sort().map(path => [path, sha(readFileSync(path))]));
  writeFileSync(output, JSON.stringify({ status: observed.ok ? "HISTORICAL_BASELINE_CAPTURED" : "HARNESS_FAILURE", observed, rows, persist,
    compiledSha256: sha(script), inputs, sourceBaseHead: execFileSync("git", ["rev-parse", "HEAD"], { encoding: "utf8" }).trim(),
    outboundCalls, realModelCalls: 0, transport: "Actual document HTTP route factory and createDocumentRuntime; local workerd/D1/R2; synthetic owner and rate limiter; analysis disabled" }, null, 2) + "\n", { flag: "wx" });
  assert.equal(response.status, 200); assert.equal(observed.ok, true, JSON.stringify(observed));
  assert.equal(rows.length, 1); assert.equal(JSON.parse(rows[0].snapshot_json).policyVersion, "2026.11.2");
  assert.equal(observed.context.preview.overlay.policyVersion, "2026.11.2");
  assert.equal(observed.context.preview.priorities.find(row => row.skillId === "sql")?.tier, 3);
  assert.deepEqual(observed.context.preview, observed.reread.preview);
  assert.equal(observed.document.text, "I have never configured SQL for production.");
  assert.equal(outboundCalls, 0);
  console.log(JSON.stringify({ status: "HISTORICAL_BASELINE_CAPTURED", contextVersionId: observed.context.contextVersionId, policy: "2026.11.2", persist, output }));
