import assert from "node:assert/strict";
import { createHash } from "node:crypto";
import { readFileSync, writeFileSync } from "node:fs";
import { build } from "esbuild";
import { Miniflare, Log, LogLevel } from "miniflare";
const [output, expectedPolicy] = process.argv.slice(2);
assert.match(output ?? "", /^work\/grounded-verification\/experience-priority-history-read-[a-z0-9-]+-results\.json$/u);
assert.ok(["2026.11.2", "2026.11.3"].includes(expectedPolicy));
const sha = value => createHash("sha256").update(value).digest("hex");
const baselinePath = "work/grounded-verification/experience-priority-history-create-20261108-b-results.json";
const bytes = readFileSync(baselinePath);
assert.equal(sha(bytes), "cce394baf72cf455b5e11927302ef2aa61c674780842139510db4762026e1f3d");
const baseline = JSON.parse(bytes);
assert.equal(sha(readFileSync("work/grounded-verification/experience-priority-native-worker-20261108.ts")), "41d38adafe774478dc68f2af553aadc568042a78e07cca1040a08a296ba20818");
const compiled = await build({ stdin: { resolveDir: process.cwd(), loader: "ts", contents: `
import worker from './work/grounded-verification/experience-priority-native-worker-20261108';
import {MATERIAL_PLANNING_POLICY_VERSION} from './app/lib/materials/planning-priority';
export default {async fetch(request,env) {const response=await worker.fetch(request,env);return Response.json({...await response.json(),currentPolicy:MATERIAL_PLANNING_POLICY_VERSION});}};
` }, bundle: true, write: false, format: "esm", platform: "browser", target: "es2022", external: ["cloudflare:workers", "node:*"], metafile: true });
let outboundCalls = 0;
const script = compiled.outputFiles[0].text, persist = baseline.persist;
const runtime = new Miniflare({ cf: false, modules: true, script, compatibilityDate: "2026-05-15", compatibilityFlags: ["nodejs_compat"],
  d1Databases: { DB: "experience-priority-history" }, d1Persist: persist + "/d1", r2Buckets: { DOCS: "experience-priority-history-docs" }, r2Persist: persist + "/r2",
  log: new Log(LogLevel.ERROR), outboundService: async () => { outboundCalls++; throw Error("Outbound denied"); } });
try {
  const db = await runtime.getD1Database("DB");
  const beforeRows = (await db.prepare("SELECT * FROM material_context_versions ORDER BY id").all()).results;
  const response = await runtime.dispatchFetch("http://localhost/history", { method: "POST", body: JSON.stringify({ action: "read", contextVersionId: baseline.observed.context.contextVersionId }) });
  const observed = await response.json();
  const afterRows = (await db.prepare("SELECT * FROM material_context_versions ORDER BY id").all()).results;
  const inputs = Object.fromEntries(Object.keys(compiled.metafile.inputs).filter(path => path !== "<stdin>").sort().map(path => [path, sha(readFileSync(path))]));
  const matches = observed.ok && JSON.stringify(observed.value) === JSON.stringify(baseline.observed.reread)
    && JSON.stringify(beforeRows) === JSON.stringify(baseline.rows) && JSON.stringify(afterRows) === JSON.stringify(beforeRows);
  writeFileSync(output, JSON.stringify({ status: matches && observed.currentPolicy === expectedPolicy ? "PASS" : "HISTORY_MISMATCH",
    observed, beforeRows, afterRows, baselinePath, baselineSha256: sha(bytes), expectedPolicy, inputs, compiledSha256: sha(script),
    outboundCalls, realModelCalls: 0 }, null, 2) + "\n", { flag: "wx" });
  assert.equal(response.status, 200); assert.equal(observed.ok, true, JSON.stringify(observed));
  assert.equal(observed.currentPolicy, expectedPolicy); assert.deepEqual(observed.value, baseline.observed.reread);
  assert.deepEqual(beforeRows, baseline.rows); assert.deepEqual(afterRows, beforeRows); assert.equal(outboundCalls, 0);
  console.log(JSON.stringify({ status: "PASS", currentPolicy: observed.currentPolicy, savedPolicy: observed.value.preview.overlay.policyVersion, persistedRowsUnchanged: true, output }));
} finally { await runtime.dispose(); }
