it.each(denied)("independent denied mention stays separate: %s (%s)", (quote, term, original) => {
  expect(mention(quote, term)).toBe(original);
  expect(groundMention(quote, [term])).toEqual([original]);

  expect(groundEvidence(quote, [term])).toEqual([]);

it("independently preserves mixed clause polarity and spelling ambiguity", () => {
  const quote = "I never used MapKit, but I used StarDB.";
  expect(groundMention(quote, ["MapKit", "StarDB"])).toEqual(["MapKit", "StarDB"]);
  expect(groundEvidence(quote, ["MapKit", "StarDB"])).toEqual(["StarDB"]);
  expect(mention("I never used MapKit; I use ＭａｐＫｉｔ.", "MapKit")).toBeNull();
  expect(groundMention("I never used MapKit; I use ＭａｐＫｉｔ.", ["MapKit"])).toEqual([]);
  expect(evidence("I never used MapKit; I use ＭａｐＫｉｔ.", "MapKit")).toBe("ＭａｐＫｉｔ");
});

it("independently binds literal nominations before reviewed normalization", () => {
  const quote = "I have not cleansed the duplicate order rows.";
  expect(mention(quote, "data cleaning")).toBe("cleansed the duplicate order rows");
  expect(groundMention(quote, ["data cleaning"])).toEqual([]);
  expect(groundMention(quote, ["cleansed the duplicate order rows"])).toEqual(["data cleaning"]);
  expect(groundMention(quote, [])).toEqual([]);
  expect(evidence(quote, "data cleaning")).toBeNull();
  expect(groundEvidence(quote, ["cleansed the duplicate order rows"])).toEqual([]);
  expect(mention("I have not cleansed data; I do data cleansing.", "data cleaning")).toBeNull();
  expect(groundMention("I have not cleansed data; I do data cleansing.", ["cleansed data"])).toEqual([]);
});

it("independently preserves nomination and original UTF-16 bounds", () => {
  const exact = "a\u0301".repeat(50), above = "a\u0301".repeat(51);
  expect(mention(exact, "á".repeat(50))).toBe(exact);
  expect(groundMention(exact, ["á".repeat(50)])).toEqual([exact]);
  expect(mention(above, "á".repeat(51))).toBeNull();
  expect(groundMention(above, ["á".repeat(51)])).toEqual([]);
  expect(groundMention("MapKit", ["", "!!!", "absent", " ", "missing", "wrong", "no MapKit", "not know MapKit", "MapKit"])).toEqual([]);
  expect(groundMention("MapKit", ["MapKit", "MapKit"])).toEqual(["MapKit"]);
  for (const label of ["not know MapKit", "不会MapKit", "NEVER USED MapKit", "尚未使用MapKit"]) {
    expect(mention(label, label)).toBeNull();
    expect(groundMention(label, [label])).toEqual([]);
    expect(evidence(label, label)).toBeNull();
    expect(groundEvidence(label, [label])).toEqual([]);
  }
});
