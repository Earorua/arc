function parsed(path) {
  const text = readFileSync(path, "utf8");
  const file = ts.createSourceFile(path, text, ts.ScriptTarget.Latest, true);
  assert.equal(file.parseDiagnostics.length, 0, path);
  const entries = file.statements.filter(node => !ts.isImportDeclaration(node)).map(node => ({
    name: ts.isVariableStatement(node) ? node.declarationList.declarations[0].name.getText(file) : node.name.getText(file),
    body: node.getText(file).replace(/^export /u, ""), node,
  }));
  return { file, text, entries };
}
