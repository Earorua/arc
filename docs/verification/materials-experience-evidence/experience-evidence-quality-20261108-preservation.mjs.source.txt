function declarations(path, text = readFileSync(path, "utf8")) {
  const sourceFile = ts.createSourceFile(path, text, ts.ScriptTarget.Latest, true);
  assert.equal(sourceFile.parseDiagnostics.length, 0, path);
  return sourceFile.statements.filter(statement => !ts.isImportDeclaration(statement)).map(statement => ({
    name: ts.isVariableStatement(statement)
      ? statement.declarationList.declarations[0].name.getText(sourceFile)
      : statement.name.getText(sourceFile),
    body: statement.getText(sourceFile).replace(/^export /u, ""),
  }));
}
const original = declarations(baselinePath, baseline);
const moved = declarations("app/lib/materials/personal-assertion-scopes.ts");
const retained = declarations("app/lib/materials/self-assessment-evidence.ts");
const current = [...moved, ...retained];
assert.equal(original.length, 28);
assert.equal(moved.length, 18);
assert.equal(current.length, original.length);
const declarationChecks = original.map(before => {
  const matches = current.filter(after => after.name === before.name);
  assert.equal(matches.length, 1, before.name);
  assert.equal(matches[0].body, before.body, before.name);
  return { name: before.name, sha256: sha(before.body), identical: true, moved: moved.some(entry => entry.name === before.name) };
});
