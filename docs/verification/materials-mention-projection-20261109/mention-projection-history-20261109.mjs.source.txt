import assert from "node:assert/strict";
import { createHash } from "node:crypto";
import { readFileSync, writeFileSync, existsSync, readdirSync, mkdtempSync } from "node:fs";
import { resolve } from "node:path";
import { build } from "esbuild";
import { Miniflare, Log, LogLevel } from "miniflare";
const [sourceRoot, output, baseline] = process.argv.slice(2);
assert.ok(sourceRoot && output);
assert.match(output, /^work\/grounded-verification\/mention-projection-history-[a-z0-9-]+-results\.json$/u);
assert.equal(existsSync(output), false);
const sha = value => createHash("sha256").update(value).digest("hex");
const entry = "work/grounded-verification/mention-projection-history-worker-20261109.ts";
const contents = readFileSync(entry, "utf8").replaceAll('"../../', '"' + resolve(sourceRoot).replaceAll("\\", "/") + "/");
const compilation = await build({ stdin: { contents, resolveDir: resolve("."), sourcefile: entry, loader: "ts" }, bundle: true, write: false,
  format: "esm", platform: "browser", target: "es2022", external: ["cloudflare:workers", "node:*"], metafile: true });
const script = compilation.outputFiles[0].text;
const prior = baseline ? JSON.parse(readFileSync(baseline, "utf8")) : null;
if (prior) assert.equal(prior.entrySha256, sha(readFileSync(entry)));
const persist = prior?.persist ?? mkdtempSync(resolve("work/grounded-verification/mention-projection-history-store-20261109-"));
let outboundCalls = 0;
const runtime = new Miniflare({ cf: false, modules: true, script, compatibilityDate: "2026-05-15", compatibilityFlags: ["nodejs_compat"],
  d1Databases: { DB: "mention-history" }, d1Persist: persist + "/d1", r2Buckets: { DOCS: "mention-history-docs" }, r2Persist: persist + "/r2",
  log: new Log(LogLevel.ERROR), outboundService: async () => { outboundCalls++; throw Error("Outbound denied"); } });
try {
  const db = await runtime.getD1Database("DB");
  if (!prior) {
    for (const file of readdirSync(resolve(sourceRoot, "drizzle")).filter(name => /^\d{4}_.+\.sql$/u.test(name) && Number(name.slice(0, 4)) <= 15).sort()) {
      for (const sql of readFileSync(resolve(sourceRoot, "drizzle", file), "utf8").split("--> statement-breakpoint").map(v => v.trim()).filter(Boolean)) await db.prepare(sql).run();
    }
    await db.prepare("INSERT INTO users(id,name,email,email_verified) VALUES('owner-a','Synthetic owner','owner-a@example.test',0)").run();
  }
  async function rows() {
    const result = {};
    for (const table of ["documents", "document_versions", "document_text_versions", "document_jobs", "document_processing_consents", "document_extractions", "document_review_drafts", "document_mutations"]) {
      const data = (await db.prepare(`SELECT * FROM ${table}`).all()).results;
      result[table] = data.sort((a, b) => JSON.stringify(a).localeCompare(JSON.stringify(b), "en"));
    }
    return result;
  }
  const before = prior ? await rows() : null;
  const response = await runtime.dispatchFetch("http://localhost/history", { method: "POST", body: JSON.stringify(prior ? { action: "read", records: prior.observed.records } : { action: "create" }) });
  const observed = await response.json();
  const after = await rows();
  const inputs = Object.fromEntries(Object.keys(compilation.metafile.inputs).filter(path => path !== entry).sort().map(path => [path, sha(readFileSync(path))]));
  const result = { status: observed.ok ? (prior ? "READ_ONLY_HISTORY_MATCHED" : "HISTORICAL_REVIEW_BASELINE_CAPTURED") : "HARNESS_FAILURE", sourceRoot: resolve(sourceRoot),
    entrySha256: sha(readFileSync(entry)), compiledSha256: sha(script), inputs, persist, observed, rows: after, outboundCalls, realModelCalls: 0,
    fixtureBoundary: "Actual DocumentsClient → HTTP factory → runtime and ReviewService → local workerd/D1/R2. Synthetic settled job/consent seeded only for result delivery. Analysis disabled; no model qualification or inference." };
  writeFileSync(output, JSON.stringify(result, null, 2) + "\n", { flag: "wx" });
  assert.equal(response.status, 200); assert.equal(observed.ok, true, JSON.stringify(observed));
  assert.equal(observed.records.length, 6); assert.equal(outboundCalls, 0);
  if (prior) {
    assert.deepEqual(observed.observed, prior.observed.observed);
    assert.deepEqual(before, prior.rows); assert.deepEqual(after, before);
  }
  console.log(JSON.stringify({ status: result.status, records: observed.records.length, persist, output, realModelCalls: 0 }));
} finally { await runtime.dispose(); }
