async function probe(spec) {
  let now = 0, sequence = 0, signed = 0, readCalls = 0, verifyCalls = 0, nonceCalls = 0;
  let completed = false, result, failure, lateRelease;
  const timers = new Map(), requests = [], events = [];
  const mode = spec.mode ?? "start", window = mode === "start" ? 10_000 : 5_000;
  const schedule = (callback, delay, kind = "watchdog") => {
    const id = ++sequence;
    timers.set(id, { callback, at: now + delay, kind });
    events.push({ kind: "scheduled", timer: kind, at: now + delay });
    return id;
  };
  const hold = () => new Promise(resolve => schedule(resolve, window + 5, "late-completion"));
  const drain = async () => { for (let index = 0; index < 32; index++) await Promise.resolve(); };
  const accepted = { scenarioId: "long", state: "researching", jobId: "synthetic-quality-job",
    requestHash: "a".repeat(64), acceptedAt: 1000, backgroundAccepted: true };
  const response = (status = 200) => ({ status, headers: new Headers(), body: accepted });
  const reject = code => { throw Object.assign(Error("Smoke driver refused"), { safeCode: code }); };
  const responseNonces = () => ({ claim: async () => {
    nonceCalls++;
    if (spec.behavior === "late-nonce") await hold();
    return true;
  } });
  const control = new Function("performance", "setTimeout", "clearTimeout", "responseNonces", "reject",
    `const hex = /^[0-9a-f]{64}$/u;\n${safeStatusSource}\n${controlSource}\nreturn control;`)(
    { now: () => now }, schedule, id => timers.delete(id), responseNonces, reject);
  const helper = {
    SCENARIOS: [{ id: "long", jobId: accepted.jobId }],
    signBridgeMessage: async () => {
      const costs = spec.signCosts ?? [0.25];
      now += costs[Math.min(signed, costs.length - 1)];
      signed++;
      return new Headers();
    },
    readBridgeBody: async message => {
      readCalls++;
      if (spec.behavior === "early-body-failure") { now += 250; throw Error("early body failure"); }
      if (spec.behavior === "late-body") await hold();
      return new TextEncoder().encode(JSON.stringify(message.body));
    },
    verifyBridgeMessage: async options => {
      verifyCalls++;
      if (spec.behavior === "late-authentication") await hold();
      await options.nonces.claim({});
    },
  };
  const transport = request => {
    requests.push(request);
    events.push({ kind: "dispatch", at: now, path: new URL(request.url).pathname });
    assert.equal(request.redirect, "error");
    assert.equal(new URL(request.url).origin, "https://smoke.example.test");
    if (spec.behavior === "sync-failure") throw Error("ambiguous");
    if (spec.behavior === "async-failure") return Promise.reject(Error("ambiguous"));
    if (spec.behavior === "hang") return new Promise(() => {});
    if (spec.behavior === "late-transport") return new Promise(resolve => { lateRelease = () => resolve(response()); });
    return Promise.resolve(response(spec.behavior === "server-retry" && requests.length === 1 ? 503 : 200));
  };
  const pending = control({ mode, scenarioId: "long" },
    { inventory: { controlOrigin: "https://smoke.example.test", controlKeyId: "synthetic-quality" } },
    helper, new Uint8Array(32), transport, () => {}).then(
      value => { result = value; completed = true; },
      error => { failure = error; completed = true; });
  for (let turn = 0; !completed && turn < 12; turn++) {
    await drain();
    if (completed) break;
    const next = [...timers.entries()].sort((a, b) => a[1].at - b[1].at)[0];
    assert.ok(next, `${spec.id}: no timer for unresolved probe`);
    const [id, timer] = next;
    timers.delete(id); now = timer.at;
    events.push({ kind: "fired", timer: timer.kind, at: now });
    timer.callback();
  }
  assert.equal(completed, true, `${spec.id}: bounded completion`);
  await pending;
  if (spec.error) assert.equal(failure?.safeCode, spec.error, spec.id);
  else {
    assert.equal(failure, undefined, spec.id);
    assert.equal(result.status, spec.status ?? "pending", spec.id);
    assert.equal(result.accepted, spec.status === "accepted", spec.id);
  }
  if (lateRelease) {
    lateRelease(); await drain();
    assert.equal(readCalls, 0, `${spec.id}: late response cannot resume body parsing`);
    assert.equal(verifyCalls, 0, `${spec.id}: late response cannot resume authentication`);
  }
  assert.equal(requests.length, spec.calls, spec.id);
  assert.deepEqual(requests.map(request => new URL(request.url).pathname),
    Array.from({ length: spec.calls }, (_, index) => `/_smoke/${mode === "status" || index ? "status" : "start"}`), spec.id);
  assert.equal(requests.every(request => request.signal.aborted), true, `${spec.id}: dispatched signal cleanup`);
  assert.equal(timers.size, 0, `${spec.id}: timer cleanup`);
  if (spec.behavior.startsWith("late-")) assert.notEqual(result?.accepted, true, `${spec.id}: cannot claim acceptance`);
  return { id: spec.id, status: "PASS", elapsedSyntheticMs: now, signed, calls: requests.length,
    readCalls, verifyCalls, nonceCalls, result, safeFailure: failure?.safeCode,
    allSignalsAborted: true, remainingTimers: timers.size, events };
}
