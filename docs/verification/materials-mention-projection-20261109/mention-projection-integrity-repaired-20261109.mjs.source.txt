import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { execFileSync } from 'node:child_process';
import { readFileSync, writeFileSync } from 'node:fs';
const output = process.argv[2];
assert.match(output ?? '', /^work\/grounded-verification\/mention-projection-integrity-[a-z0-9-]+-results\.json$/u);
const sha = value => createHash('sha256').update(value).digest('hex');
const git = (...args) => execFileSync('git', args, { encoding: 'utf8', maxBuffer: 32 * 1024 * 1024 });
const files = [...new Set(git('ls-files', '--cached', '--others', '--exclude-standard', '-z').split('\0').filter(path => path && !path.startsWith('docs/')))].sort();
const filesSha256 = Object.fromEntries(files.map(path => [path, sha(readFileSync(path))]));
const beforePath = 'work/grounded-verification/experience-priority-snapshot-20261109-a-results.json';
const beforeBytes = readFileSync(beforePath);
assert.equal(sha(beforeBytes), '0ef2c32dada1182b8ddda84518f3e19a6157593325e37712dcbaad6ff32d8ad8');
const before = JSON.parse(beforeBytes);
assert.equal(before.manifestSha256, '8ab545fbd75adeaa8274ac9c7b89d7a7bbd37085a473b748071f6d11a5042ecb');
for (const [path, digest] of Object.entries(before.filesSha256)) assert.equal(sha(readFileSync(before.checkout + '/' + path)), digest, 'Historical copy: ' + path);
const changed = files.filter(path => before.filesSha256[path] && before.filesSha256[path] !== filesSha256[path]);
const added = files.filter(path => !before.filesSha256[path]);
const removed = Object.keys(before.filesSha256).filter(path => !filesSha256[path]);
assert.deepEqual(changed, [
  'app/components/materials/material-confirmation.tsx', 'app/contracts/documents.ts', 'app/lib/materials/skill-labels.ts',
  'app/server/documents/extraction-validator.ts', 'app/server/documents/policy.ts', 'app/server/documents/review-service.ts',
  'tests/components/material-confirmation-source.test.tsx', 'tests/components/material-source-term-correction.test.tsx',
  'tests/server/document-extraction.test.ts', 'tests/server/document-review-draft.test.ts',
  'tests/server/document-source-span-provider.test.ts', 'worker/document-executor/source-span-extraction.ts',
  'scripts/test-research-cloud-smoke.mjs', 'tests/server/research-cloud-smoke-composition.test.ts',
].sort());
assert.deepEqual(added, ['tests/helpers/material-mention-review-fixture.ts', 'tests/lib/materials/reviewed-source-mentions.test.ts', 'tests/server/material-mention-review-flow.test.ts'].sort());
assert.deepEqual(removed, []);
const previousPath = 'docs/verification/materials-experience-evidence/experience-evidence-root-integrity-20261108-quoted-b-results.json';
const previousBytes = readFileSync(previousPath);
assert.equal(sha(previousBytes), '6588362a66d6b737212d6f801f8cb2109d81fa21b54374506ec62cd96799809c');
const previous = JSON.parse(previousBytes);
for (const [path, digest] of Object.entries(previous.frozenReferenceSha256)) assert.equal(sha(readFileSync(path)), digest, path);
assert.equal(sha(readFileSync(previous.approvedSpec.path)), previous.approvedSpec.sha256);
const capturePath = 'work/grounded-verification/edited-assessment-course-capture-20261108-results.json';
const capture = JSON.parse(readFileSync(capturePath));
const dependencies = capture.dependencies.map(dependency => ({ path: dependency.path, before: dependency.sha256, current: sha(readFileSync(dependency.path)) }));
assert.equal(dependencies.length, 85);
assert.deepEqual(dependencies.filter(row => row.before !== row.current), [], 'Captured course dependency graph remains unchanged');
const snapshotPath = 'work/grounded-verification/mention-projection-snapshot-20261109-b-results.json';
const snapshotBytes = readFileSync(snapshotPath), snapshot = JSON.parse(snapshotBytes);
assert.deepEqual(filesSha256, snapshot.filesSha256);
for (const [path, digest] of Object.entries(filesSha256)) assert.equal(sha(readFileSync(snapshot.checkout + '/' + path)), digest, path);
const result = { status: 'PASS', sourceBaseAtCapture: git('rev-parse', 'HEAD').trim(), sourceFileCount: files.length,
  sourceManifestSha256: sha(JSON.stringify(filesSha256)), filesSha256, beforePath, beforeSha256: sha(beforeBytes),
  previousPath, previousSha256: sha(previousBytes), changed, added, removed, snapshotPath, snapshotSha256: sha(snapshotBytes), checkout: snapshot.checkout,
  frozenReferenceCount: Object.keys(previous.frozenReferenceSha256).length, frozenReferenceSha256: previous.frozenReferenceSha256,
  approvedSpec: previous.approvedSpec, courseCapturePath: capturePath, courseDependencyCount: dependencies.length,
  courseDependencyBytesUnchanged: true, newCourseReferenceApproved: false, realModelCalls: 0 };
writeFileSync(output, JSON.stringify(result, null, 2) + '\n', { flag: 'wx' });
console.log(JSON.stringify({ status: 'PASS', files: files.length, sourceManifestSha256: result.sourceManifestSha256,
  changed, added, frozenReferences: result.frozenReferenceCount, courseDependencies: dependencies.length, output }));
