import { createDocumentRuntime } from "../../app/server/documents/runtime";
import { createDocumentRoute } from "../../app/server/http/document-route-factories";
import { documentEndpointContracts, type DocumentEndpoint } from "../../app/contracts/documents-api";
import { createDocumentsClient } from "../../app/lib/documents-client";
import { D1DocumentRepository } from "../../app/server/documents/d1-repository";
import { DocumentStorage } from "../../app/server/documents/storage";
import { ReviewService } from "../../app/server/documents/review-service";

const clock = () => new Date("2026-11-09T01:00:00.000Z");
export default { async fetch(request: Request, env: { DB: D1Database; DOCS: R2Bucket }) {
  const payload = await request.json() as { action: "create" | "read"; records?: any[] };
  const trace: unknown[] = [];
  let serial = 0;
  const id = () => "mention-history-" + ++serial;
  const runtime = createDocumentRuntime({ DB: env.DB, DOCUMENT_ASSETS: env.DOCS, ARC_DOCUMENTS_ENABLED: "true", ARC_DOCUMENT_PLANNING_ENABLED: "true", ARC_DOCUMENT_ANALYSIS_ENABLED: "false" }, { now: clock, createId: id });
  const review = new ReviewService(new D1DocumentRepository(env.DB, clock, id), new DocumentStorage(env.DOCS));
  const client = createDocumentsClient({ fetch: async (input, init) => {
    const url = new URL(String(input), "https://arc.test");
    const matched = Object.entries(documentEndpointContracts).find(([, contract]) => contract.method === init?.method && new RegExp("^" + contract.path.replace(/\[[A-Za-z]+\]/gu, "([^/]+)") + "$").test(url.pathname));
    if (!matched) throw Error("Unmatched route " + url.pathname);
    const [endpoint, contract] = matched, params: Record<string, string> = {};
    const keys = [...contract.path.matchAll(/\[([A-Za-z]+)\]/gu)].map(match => match[1]);
    const values = url.pathname.match(new RegExp("^" + contract.path.replace(/\[[A-Za-z]+\]/gu, "([^/]+)") + "$"))!;
    keys.forEach((key, index) => { params[key] = decodeURIComponent(values[index + 1]); });
    const headers = new Headers(init?.headers); if (init?.method !== "GET") headers.set("origin", "https://arc.test");
    const response = await createDocumentRoute(endpoint as DocumentEndpoint, { requireUser: async () => ({ id: "owner-a" }), configuredOrigin: () => "https://arc.test", createService: () => runtime,
      rateLimiter: { reserve: async () => ({ allowed: true }) } })(new Request(url, { ...init, headers }), { params });
    trace.push({ endpoint, status: response.status });
    return response;
  } });
  try {
    const records = payload.records ?? [];
    if (payload.action === "create") {
      const rows = [
        ["v1", "I have never configured SQL for production.", ["SQL"], null],
        ["v1", "我在指导下清洗数据。", ["data cleaning"], "guided"],
        ["v2", "I have never configured SQL for production.", ["SQL"], null],
        ["v2", "我在指导下清洗数据。", ["data cleaning"], "guided"],
        ["v2", "I do not know SQL.", ["SQL"], null],
        ["v2", "I configured React under supervision.", [], null],
      ] as const;
      for (const [index, [version, text, labels, hint]] of rows.entries()) {
        const providerPolicyId = "materials-openrouter-kimi-k31-pro-mxfp4-zdr-extractive-" + version;
        const created = await client.request("createDocument", { kind: "resume", displayName: `Synthetic old ${version} ${index}`, text, mutationId: `create-${index}` });
        const consentId = `consent-${index}`, jobId = `job-${index}`;
        // Local result-delivery fixture. These rows never dispatch a job, hold no real credential, and incur no inference.
        await env.DB.prepare("INSERT INTO document_processing_consents VALUES(?,?,?,?,?,'analysis','2026.11.1',?,0,0,NULL)")
          .bind("owner-a", consentId, created.textVersionId, created.documentId, created.documentVersionId, providerPolicyId).run();
        await env.DB.prepare("INSERT INTO document_jobs(owner_id,id,text_version_id,document_id,document_version_id,consent_id,generation,document_revision,revision,state,dispatch_state,execution_slot,queued_at,queue_expires_at,provider_policy_id,price_version,policy_version,reservation_id,updated_at) VALUES(?,?,?,?,?,?,0,0,0,'running','settled',NULL,0,1,?,'2026.11.1','2026.11.1','offline-reservation',0)")
          .bind("owner-a", jobId, created.textVersionId, created.documentId, created.documentVersionId, consentId, providerPolicyId).run();
        const result = await review.acceptAiExtraction("owner-a", { ...created, jobId, consentId, generation: 0, documentRevision: 0,
          candidate: { schemaVersion: "2026.11.1", items: [{ id: "fact", kind: "experience", text, requirementLevel: null,
            skillLabels: [...labels], selfLevelHint: hint, source: { segmentId: "segment-1", quote: text } }], reviewPoints: [] } });
        if (result.disposition !== "initialized") throw Error("History delivery rejected");
        const initial = (await client.request("readDocumentReviewDraft", { documentId: created.documentId, textVersionId: created.textVersionId })).draft!;
        const saved = await client.request("saveDocumentReviewDraft", { documentId: created.documentId, textVersionId: created.textVersionId, extractionId: initial.extractionId,
          items: initial.items, selfAssessmentHints: initial.selfAssessmentHints, reviewPoints: initial.reviewPoints, expectedRevision: initial.revision, mutationId: `save-${index}` });
        records.push({ created, extractionId: result.extractionId, providerPolicyId, initial, saved });
      }
    }
    const observed = [];
    for (const record of records) {
      const { created, extractionId } = record;
      observed.push({ record,
        document: await client.request("readDocument", { documentId: created.documentId }),
        review: await client.request("readDocumentReviewDraft", { documentId: created.documentId, textVersionId: created.textVersionId }),
        extraction: await client.request("readDocumentExtraction", { documentId: created.documentId, extractionId }),
      });
    }
    return Response.json({ ok: true, records, observed, trace });
  } catch (error) { return Response.json({ ok: false, message: String(error instanceof Error ? error.message : error), trace }); }
} };
