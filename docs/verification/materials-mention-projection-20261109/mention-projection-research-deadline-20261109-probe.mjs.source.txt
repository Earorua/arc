async function probe({ id, signCosts, behavior, expectedCalls, expectedStatus }) {
  let now = 0, nextTimer = 0, signed = 0, completed = false, result, error;
  const timers = new Map(), events = [], requests = [];
  const setTimer = (callback, delay) => {
    const timer = ++nextTimer;
    timers.set(timer, { at: now + delay, callback });
    events.push({ event: "timer-set", now, delay, at: now + delay, timer });
    return timer;
  };
  const clearTimer = timer => { timers.delete(timer); };
  const reject = code => { throw Object.assign(Error("Smoke driver refused"), { safeCode: code }); };
  const responseNonces = () => ({ claim: async (_claim, check) => { check(); return true; } });
  const control = new Function("performance", "setTimeout", "clearTimeout", "responseNonces", "reject",
    "Request", "URL", "TextEncoder", "TextDecoder", "AbortController", "Date",
    `const hex = /^[0-9a-f]{64}$/u;\n${safeStatusSource}\n${controlSource}\nreturn control;`)(
    { now: () => now }, setTimer, clearTimer, responseNonces, reject,
    Request, URL, TextEncoder, TextDecoder, AbortController, Date);
  const helper = {
    SCENARIOS: [{ id: "long", jobId: accepted.jobId }],
    signBridgeMessage: async () => {
      now += signCosts[Math.min(signed, signCosts.length - 1)];
      events.push({ event: "signed", now, call: ++signed });
      return {};
    },
    readBridgeBody: async response => new TextEncoder().encode(JSON.stringify(response.probeBody)),
    verifyBridgeMessage: async options => { await options.nonces.claim({}); },
  };
  const transport = request => {
    requests.push(request);
    events.push({ event: "transport", now, path: new URL(request.url).pathname });
    if (behavior === "hang") return new Promise(() => {});
    if (behavior === "ambiguous") return Promise.reject(Error("Synthetic ambiguous result"));
    const probeBody = behavior === "reconcile" && requests.length === 1
      ? { scenarioId: "long", state: "initializing", backgroundAccepted: false }
      : accepted;
    return Promise.resolve({ status: 200, headers: {}, probeBody });
  };
  const pending = control({ mode: "start", scenarioId: "long" },
    { inventory: { controlOrigin: "https://smoke.example.test", controlKeyId: "synthetic-control" } },
    helper, new Uint8Array(32), transport, () => {}).then(
      value => { result = value; completed = true; },
      failure => { error = failure; completed = true; });
  for (let turn = 0; !completed && turn < 20; turn++) {
    // Drain immediate signing/transport/body/nonce continuations before advancing time.
    for (let microtask = 0; microtask < 32; microtask++) await Promise.resolve();
    if (completed) break;
    const next = [...timers.entries()].sort((a, b) => a[1].at - b[1].at)[0];
    assert.ok(next, `${id}: unresolved operation has no scheduled timer`);
    const [timer, task] = next;
    timers.delete(timer); now = task.at;
    events.push({ event: "timer-fired", now, timer });
    task.callback();
  }
  assert.ok(completed, `${id}: probe must complete within the local deterministic bound`);
  await pending;
  if (error) throw error;
  assert.equal(result.status, expectedStatus);
  assert.equal(result.accepted, expectedStatus === "accepted");
  assert.ok(requests.every(request => request.signal.aborted), `${id}: all dispatched signals aborted`);
  let callContract;
  try { assert.equal(requests.length, expectedCalls); callContract = { status: "PASS" }; }
  catch (failure) { callContract = { status: "RED", actual: failure.actual, expected: failure.expected, message: failure.message }; }
  return { id, fakeElapsedMs: now, expectedCalls, actualCalls: requests.length, result, callContract,
    paths: requests.map(request => new URL(request.url).pathname), allSignalsAborted: true, events };
}
