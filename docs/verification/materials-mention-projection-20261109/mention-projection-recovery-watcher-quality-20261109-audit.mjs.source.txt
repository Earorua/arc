  identities[path] = actual;

}

const runs = {};
function retainedRun(label, expectedExit, expectedHash) {
  const path = `${work}${label}.json`;
  const report = identified(path);
  const logPath = `${work}${label}.log`;
  const logBytes = bytes(logPath);
  assert.equal(report.exit, expectedExit);
  assert.equal(report.nonLoopbackDenials, 0);
  assert.equal(hash(logBytes), report.logSha256);
  if (expectedHash) assert.equal(report.logSha256, expectedHash);
  identities[logPath] = report.logSha256;
  runs[label] = { ...report, reportPath: path, reportSha256: identities[path], logPath };
  return stripVTControlCharacters(logBytes.toString());
}
