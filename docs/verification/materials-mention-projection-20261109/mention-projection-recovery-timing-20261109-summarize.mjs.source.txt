const sha256 = value => createHash("sha256").update(value).digest("hex");
const readJSON = name => JSON.parse(readFileSync(join(work, name), "utf8"));
const first = readJSON(`${prefix}-prepare-results.json`);
const b = readJSON(`${prefix}-prepare-b-results.json`);
const reversal = readJSON(`${prefix}-prepare-reversal-results.json`);
const round = n => Math.round(n * 1000) / 1000;

function invertOne(text, before, after) {
  assert.equal(text.split(after).length - 1, 1, "Inverse edit must match uniquely");
  return text.replace(after, before);
}
function proveCopy(name, manifest, subsequent = []) {
  const file = manifest.files?.[name] ?? manifest.copies[name];
  const bytes = readFileSync(file.path);
  assert.equal(sha256(bytes), file.sha256, "Probe bytes unchanged after execution");
  let text = bytes.toString("utf8");
  for (const edit of [...subsequent].reverse()) text = invertOne(text, edit.before, edit.after);
  for (const edit of [...first.edits].reverse()) if (edit.name === name) text = invertOne(text, edit.before, edit.after);
  assert.equal(sha256(text), first.originals[name].sha256, "All changes accounted for exactly");
  assert.equal(sha256(readFileSync(first.originals[name].path)), first.originals[name].sha256, "Original source preserved");
  return { copySha256: file.sha256, originalSha256: first.originals[name].sha256, inverseEditsRecoverExactOriginal: true };
}

  assert.equal(sha256(bytes), result.logSha256, "Raw wrapper log must retain hash");
  const records = bytes.toString("utf8").split(/\r?\n/u).flatMap(line => {
    try { return [JSON.parse(line)]; } catch { return []; }
  });
  const marks = records.filter(record => record.event === "offline-recovery-timing" && !record.heartbeat);
  const find = phase => marks.find(record => record.phase === phase);
  const phaseMs = Object.fromEntries(phases.map(phase => [phase, round(find(`${phase}:after`).elapsedMs - find(`${phase}:before`).elapsedMs)]));
  const aborted = find("test:abort");
  const bodyStart = find("test:startResearch:before");
  const allAssertions = find("test:allAssertions:after");
  runs[label] = {
    ...result,
    logPath: join(work, `${name}.log`),
    resultPath: join(work, `${name}.json`),
    passed: /ℹ pass 1(?:\r?\n)/u.test(bytes.toString("utf8")),

  };
