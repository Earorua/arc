async function setup(adopted: boolean) {
  const db = createMaterialsD1(':memory:');
  seedMaterialSelection(db);
  // Independent synthetic BEFORE input, not an expected after-state. Use a primary SQL must
  // so an additional SQL soft focus cannot change the priority tier or current order.
  db.database.prepare("UPDATE document_confirmations SET items_json=? WHERE id='confirmation-a'").run(JSON.stringify([
    { id: 'confirmed-a', kind: 'requirement', text: 'SQL', origin: 'user', sourceRef: null, requirementLevel: 'must' },
  ]));
  let id = 0;
  const sourceResolver = materialSourceResolver();
  const repository = new D1PlanningRepository(db as unknown as D1Database, { sourceResolver, now: materialNow, createId: () => `stored-${++id}` });
  const contexts = new MaterialContextService({ db: db as unknown as D1Database, sourceResolver, planningRepository: repository, now: materialNow, createId: () => `context-${++id}` });
  const planning = new PlanningService({ repository, sourceResolver, materials: contexts, now: materialNow, createId: () => `event-${++id}` });
  const documents = createDocumentServices({ db: db as unknown as D1Database, bucket: memoryBucket(), clock: materialNow, id: () => `document-${++id}` });
  const proof = new ProofService({ repository: new D1ProofRepository(db as unknown as D1Database), blueprint: flagshipBlueprint, registry: flagshipUnitRegistry, now: materialNow, createId: () => `proof-${++id}` });
  const doc = await documents.createDocument(owner, { kind: 'supplement', displayName: 'Synthetic SQL focus', text: 'SQL', mutationId: 'create-focus' });
  const item = { id: 'sql-focus', kind: 'requirement' as const, text: 'SQL', origin: 'user' as const, sourceRef: null, requirementLevel: 'unspecified' as const };
  const { draft } = await documents.saveDocumentReviewDraft(owner, { documentId: doc.documentId, textVersionId: doc.textVersionId, extractionId: null,
    items: [item], selfAssessmentHints: [], expectedRevision: 0, mutationId: 'review-focus' });
  const confirmation = await documents.createDocumentConfirmation(owner, { documentId: doc.documentId, documentVersionId: doc.documentVersionId,
    textVersionId: doc.textVersionId, items: draft.items, reviewRevision: draft.revision, expectedRevision: doc.revision, mutationId: 'confirm-focus' });
  const entries = [{ slot: 'jd' as const, documentId: 'doc-a', documentVersionId: 'version-a', confirmationId: 'confirmation-a' },
    { slot: 'supplement' as const, documentId: doc.documentId, documentVersionId: doc.documentVersionId, confirmationId: confirmation.confirmationId }];
  const learningIntents = [{ confirmationId: confirmation.confirmationId, itemId: item.id }];
  const request = materialContextRequest();
  const initialSelection = await documents.createMaterialSelection(owner, { scope: request.scope, entries, mutationId: 'initial-selection' });
  const initialContext = await contexts.create(owner.ownerId, { ...request, selectionRef: initialSelection, ...(adopted ? { learningIntents } : {}), mutationId: 'initial-context' });
  const { scope: ignoredScope, selectionRef: ignoredSelection, ...generation } = request; void ignoredScope; void ignoredSelection;
  let workspace = (await planning.generate(owner.ownerId, { ...generation, materials: { selectionRef: initialSelection, contextVersionId: initialContext.contextVersionId, previewToken: initialContext.previewToken } })).workspace;
  for (let n = 0; n < 2; n++) {
    const plan = workspace.planVersions.find(p => p.id === workspace.activePlanVersionId)!;
    const day = plan.days.find(d => d.primaryUnitId !== null)!;
    workspace = (await planning.appendEvent(owner.ownerId, { mutationId: `completion-${n}`, baseVersionId: workspace.activePlanVersionId,
      event: { kind: 'completed', unitId: day.primaryUnitId!, actualMinutes: 31 + n, planningDate: day.date } })).workspace;
  }
  const complete = workspace.events.filter(e => e.kind === 'completed').at(-1)!;
  const unit = workspace.dailyUnits.find(u => u.id === complete.unitId)!;
  await proof.create(owner.ownerId, { mutationId: 'existing-proof', baseRevision: 0, intent: 'submit', validatorKey: null,
    dailyUnitId: unit.id, title: 'Synthetic preserved artifact', kind: 'document', summary: 'Existing local test evidence.',
    artifactUrl: 'https://example.com/offline-proof', assetId: null, skillIds: [unit.skillId], completionCriteria: unit.completionCriteria, visibility: 'private' });
  const scope = { kind: 'goal' as const, goalId: workspace.goalId, workspaceId: workspace.id, baseVersionId: workspace.activePlanVersionId, revision: workspace.revision };
  const currentPlan = workspace.planVersions.find(p => p.id === workspace.activePlanVersionId)!;
  const inputs = { source: request.source, audit: workspace.audit, availability: workspace.availability, target: workspace.target,
    planningDate: currentPlan.planningDate, selectedScope: request.selectedScope };
  const selectionRef = await documents.createMaterialSelection(owner, { scope, entries, mutationId: 'goal-selection' });
  async function capture() {
    return { workspace: await planning.getWorkspace(owner.ownerId), proof: await proof.getWorkspace(owner.ownerId),
      rows: Object.fromEntries(tables.map(table => [table, db.database.prepare(`SELECT * FROM ${table} ORDER BY rowid`).all()])) };
  }
  const currentContextIds = () => (db.database.prepare("SELECT DISTINCT context_version_id FROM goal_document_links WHERE enabled=1 ORDER BY context_version_id").all() as { context_version_id: string }[]).map(row => row.context_version_id);
  return { db, documents, contexts, planning, proof, doc, confirmation, entries, learningIntents, initialContext, workspace, scope, inputs, selectionRef, capture, currentContextIds };
}

async function focusProbe(initialAdopted: boolean, nextAdopted: boolean, changeSelf = false) {
  const f = await setup(initialAdopted);
  try {
    const before = await f.capture();
    const beforeCurrentContextIds = f.currentContextIds();
    const audit = changeSelf ? { ...f.inputs.audit, answers: f.inputs.audit.answers.map(a => a.skillId === 'sql' ? { ...a, level: 'guided' as const } : a) } : f.inputs.audit;
    const preview = await f.contexts.create(owner.ownerId, { ...f.inputs, audit, scope: f.scope, selectionRef: f.selectionRef,
      ...(nextAdopted ? { learningIntents: f.learningIntents } : {}), mutationId: 'changed-context' });
    const result = await f.planning.proposeMaterialReplanResponse(owner.ownerId, { scope: f.scope, selectionRef: f.selectionRef,
      contextVersionId: preview.contextVersionId, previewToken: preview.previewToken, mutationId: 'changed-proposal' });
    const after = await f.capture();
    assert.equal(result.outcome, 'no-change');
    assert.deepEqual(after.workspace, before.workspace);
    assert.deepEqual(after.proof, before.proof);
    assert.deepEqual(f.currentContextIds(), beforeCurrentContextIds);
    return { initialAdopted, nextAdopted, changeSelf, result, beforeCurrentContextIds, afterCurrentContextIds: f.currentContextIds(),
      initialContextId: f.initialContext.contextVersionId, candidateContextId: preview.contextVersionId, preview,
      fullWorkspaceUnchanged: true, fullProofWorkspaceUnchanged: true, before, after };
  } finally { f.db.close(); }
}

async function deletionProbe() {
  const f = await setup(true);
  try {
    const before = await f.capture();
    const result = await f.documents.deleteDocument(owner, { documentId: f.doc.documentId, expectedRevision: f.confirmation.revision, mutationId: 'delete-focus' });
    const after = await f.capture();
    assert.deepEqual(after.workspace, before.workspace);
    assert.deepEqual(after.proof, before.proof);
    const document = f.db.database.prepare('SELECT id,lifecycle,revision FROM documents WHERE id=?').get(f.doc.documentId);
    return { result, document, fullWorkspaceUnchanged: true, fullProofWorkspaceUnchanged: true, before, after };
  } finally { f.db.close(); }
}

async function staleContextProbe() {
  const f = await setup(true);
  try {
    const preview = await f.contexts.create(owner.ownerId, { ...f.inputs, scope: f.scope, selectionRef: f.selectionRef, mutationId: 'old-context' });
    const plan = f.workspace.planVersions.find(p => p.id === f.workspace.activePlanVersionId)!;
    const day = plan.days.find(d => d.primaryUnitId !== null)!;
    await f.planning.appendEvent(owner.ownerId, { mutationId: 'completion-after-preview', baseVersionId: f.workspace.activePlanVersionId,
      event: { kind: 'completed', unitId: day.primaryUnitId!, actualMinutes: 44, planningDate: day.date } });
    const beforeAttempt = await f.capture();
    let code: string | null = null;
    try { await f.planning.proposeMaterialReplanResponse(owner.ownerId, { scope: f.scope, selectionRef: f.selectionRef, contextVersionId: preview.contextVersionId, previewToken: preview.previewToken, mutationId: 'stale-proposal' }); }
    catch (error) { code = errorCode(error); }
    const afterAttempt = await f.capture();
    assert.equal(code, 'CONFLICT');
    assert.deepEqual(afterAttempt, beforeAttempt);
    return { action: 'propose-material-replan-from-stale-context', code, contextId: preview.contextVersionId,
      previewBoundRevision: f.scope.revision, actualPlanningRevision: afterAttempt.workspace!.revision, beforeAttempt, afterAttempt };
  } finally { f.db.close(); }
}

async function unresolvedConflictProbe() {
  const f = await setup(false);
  try {
    const doc = await f.documents.createDocument(owner, { kind: 'supplement', displayName: 'Synthetic conflict', text: 'SQL must. SQL preferred.', mutationId: 'create-conflict' });
    const { draft } = await f.documents.saveDocumentReviewDraft(owner, { documentId: doc.documentId, textVersionId: doc.textVersionId, extractionId: null,
      items: [{ id: 'conflict-item', kind: 'requirement', text: 'SQL must. SQL preferred.', origin: 'user', sourceRef: null, requirementLevel: 'unspecified' }],
      selfAssessmentHints: [], expectedRevision: 0, mutationId: 'review-conflict' });
    const points = materialReviewPointsSchema.parse([{ id: 'unresolved-conflict', code: 'conflicting-requirement', resolution: 'unresolved',
      sourceRef: { documentId: doc.documentId, documentVersionId: doc.documentVersionId, textVersionId: doc.textVersionId, start: 0, end: 23 } }]);
    // Legitimate source-bound unresolved-review BEFORE fixture. No model and no after-state seed.
    f.db.database.prepare('UPDATE document_review_drafts SET review_points_json=? WHERE id=?').run(JSON.stringify(points), draft.id);
    let code: string | null = null;
    try { await f.documents.createDocumentConfirmation(owner, { documentId: doc.documentId, documentVersionId: doc.documentVersionId, textVersionId: doc.textVersionId,
      items: draft.items, reviewRevision: draft.revision, expectedRevision: doc.revision, mutationId: 'confirm-conflict' }); }
    catch (error) { code = errorCode(error); }
    assert.equal(code, 'invalid-content');
    const count = f.db.database.prepare('SELECT count(*) n FROM document_confirmations WHERE document_id=?').get(doc.documentId);
    assert.equal((count as { n: number }).n, 0);
    return { code, confirmationCount: count, reviewPoints: points, planningWorkspace: await f.planning.getWorkspace(owner.ownerId) };
  } finally { f.db.close(); }
}
