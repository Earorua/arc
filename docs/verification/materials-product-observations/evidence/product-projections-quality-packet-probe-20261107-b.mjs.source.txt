const sha = bytes => createHash('sha256').update(bytes).digest('hex');

function verifyReference(value) {
  if (!value || typeof value !== 'object') return;
  if (typeof value.path === 'string' && typeof value.sha256 === 'string') {
    const bytes = readFileSync(value.path);
    assert.equal(sha(bytes), value.sha256, value.path);
    if (typeof value.bytes === 'number') assert.equal(bytes.length, value.bytes, value.path + ': bytes');
    checkedReferences.set(value.path, value.sha256);
  }
  for (const child of Object.values(value)) verifyReference(child);
}
function gitBlobs(specs) {
  const bytes = execFileSync('git', ['cat-file', '--batch'], { input: specs.join('\n') + '\n', maxBuffer: 32 * 1024 * 1024, windowsHide: true });
  let cursor = 0;
  return specs.map(spec => {
    const end = bytes.indexOf(10, cursor);
    assert.ok(end >= cursor, spec + ': batch header');
    const [objectId, type, size] = bytes.subarray(cursor, end).toString('utf8').split(' ');
    assert.equal(type, 'blob', spec + ': expected Git blob');
    const length = Number(size), start = end + 1;
    assert.ok(Number.isInteger(length) && length >= 0, spec + ': blob length');
    const content = bytes.subarray(start, start + length);
    assert.equal(content.length, length, spec + ': blob bytes');
    assert.equal(bytes[start + length], 10, spec + ': separator');
    cursor = start + length + 1;
    return { objectId, content };
  });
}
