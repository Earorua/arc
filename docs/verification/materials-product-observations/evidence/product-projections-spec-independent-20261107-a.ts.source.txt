const canonical = (value: unknown): string => {
  if (Array.isArray(value)) return `[${value.map(canonical).join(",")}]`;
  if (value !== null && typeof value === "object") return `{${Object.keys(value).sort().map(key => `${JSON.stringify(key)}:${canonical((value as Record<string, unknown>)[key])}`).join(",")}}`;
  return JSON.stringify(value);
};

function projected(report: ReturnType<typeof collectProjectionObservations>, at = 0) {
  const result = report.scenarios[at]!;
  assert.equal(result.outcome, "projected");
  if (result.outcome !== "projected") throw new Error("Projection missing");
  return result;
}
function assertIncomplete(report: ReturnType<typeof collectProjectionObservations>) {

const cliResults: any[] = [];
function cli(name: string, args: string[]) {
  const result = spawnSync(process.execPath, ["--import", "tsx", collectorPath, ...args], { encoding: "utf8", windowsHide: true, timeout: 30_000, maxBuffer: 32 * 1024 * 1024 });
  assert.equal(result.error, undefined);
  cliResults.push({ name, status: result.status, stdoutBytes: Buffer.byteLength(result.stdout), stdoutSha256: hash(result.stdout), stderr: result.stderr, stderrSha256: hash(result.stderr) });
  return result;
}
function cliError(name: string, path: string, code: string) {
  const result = cli(name, [path]);
  assert.equal(result.status, 2);
  assert.equal(result.stdout, "");
  assert.equal(result.stderr, `projection-collector: ${code}\n`);
}

check("source span offsets are native, and only labels supported by the quoted span survive", () => {
  const stimulus = scenario();
  stimulus.reference.text = "😀 SQL is required. Rust elsewhere.";
  stimulus.reference.segments = [{ id: "spec-span", start: 3, end: 19 }];
  stimulus.extraction.items[0]!.text = "SQL and Rust are both required.";
  stimulus.extraction.items[0]!.skillLabels = ["SQL", "Rust"];
  const observed = projected(collect([stimulus]));
  assert.deepEqual(observed.projection.items[0]?.skillLabels, ["SQL"]);
  assert.deepEqual(observed.projection.items[0]?.sourceRef, { documentId: "spec-document", documentVersionId: "spec-version", textVersionId: "spec-text", start: 3, end: 19 });
  for (const mutation of ["split-start", "split-end", "beyond-end", "inverted", "duplicate"] as const) {
    const invalid = structuredClone(stimulus);
    if (mutation === "split-start") invalid.reference.segments[0]!.start = 1;
    if (mutation === "split-end") { invalid.reference.segments[0]!.start = 0; invalid.reference.segments[0]!.end = 1; }
    if (mutation === "beyond-end") invalid.reference.segments[0]!.end = invalid.reference.text.length + 1;
    if (mutation === "inverted") invalid.reference.segments[0]!.end = 2;
    if (mutation === "duplicate") invalid.reference.segments.push({ ...invalid.reference.segments[0]! });
    assert.throws(() => collect([invalid]), { message: "invalid-input" }, mutation);
  }
});
