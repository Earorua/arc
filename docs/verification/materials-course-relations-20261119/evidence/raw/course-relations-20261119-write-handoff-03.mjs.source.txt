const sha256 = path => createHash("sha256").update(readFileSync(path)).digest("hex");

if (sha256(previousPath) !== archive.handoffSha256 || sha256(archive.diagnosticPath) !== archive.diagnosticSha256)
  throw Error("Pinned handoff 02 or diagnostic 02 changed");
for (const file of archive.files) if (sha256(file.archivedPath) !== file.archivedSha256)
  throw Error(`Archived source changed: ${file.path}`);

  throw Error("First handoff, archive or diagnostic changed");

  throw Error(`Original archived source changed: ${file.path}`);
