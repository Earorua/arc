// @vitest-environment node
import { writeFileSync } from "node:fs";
import { performance } from "node:perf_hooks";
import { expect, it } from "vitest";
import { documentReviewDraftSchema, type DocumentReviewDraft } from "../../app/contracts/documents";
import { retainedMaterialSelfAssessmentHints as retain } from "../../app/lib/materials/skill-labels";
import { supportsMaterialSelfLevelHint as current } from "../../app/lib/materials/self-assessment-evidence";
import { supportsMaterialSelfLevelHint as base } from "./edited-assessment-quality-20261108-base-evidence";

const texts = [
  "按指导完成 SQL 查询练习",
  "按照指导完成 SQL 查询练习",
  "按指导示例完成 SQL 查询练习",
  "按照指导教程完成 SQL 查询练习",
  "按指导完成 SQL 示例",
  "按照指导完成 SQL 教程练习",
  "按指导完成不了 SQL 查询练习",
  "按照指导完成不了 SQL 查询练习",
  "在导师指导下完成不了 SQL 查询练习",
  "按指导无法完成 SQL 查询练习",
  "按照指导没有完成 SQL 查询练习",
  "按指导完成同事 SQL 查询练习的记录",
  "在导师指导下完成同事 SQL 查询练习的记录",
];

it("records bounded phrase observations with a pre-change semantic baseline", () => {
  const observations = texts.map(text => ({ text, base: base({kind: "experience", text, labels: ["SQL"], level: "guided"}),
    current: current({kind: "experience", text, labels: ["SQL"], level: "guided"}) }));
  writeFileSync("work/grounded-verification/edited-assessment-quality-20261108-phrase-observations.json", JSON.stringify(observations, null, 2) + "\n");
  console.log(JSON.stringify({phraseObservations: observations}));
  expect(observations).toHaveLength(texts.length);
});

it("records repeated full-size retained-hint evaluation within the existing wire limits", () => {
  const text = ("I tested SQL under supervision. " + "Supporting context ".repeat(60)).slice(0, 1000);
  const sourceRef = {documentId: "d", documentVersionId: "v", textVersionId: "t", start: 0, end: 1000};
  const draft: DocumentReviewDraft = {id: "r", schemaVersion: "2026.11.1", documentId: "d", documentVersionId: "v", textVersionId: "t",
    extractionId: "e", origin: "mixed", revision: 1, updatedAt: "2026-11-08T00:00:00.000Z", reviewProjection: "grounded-skill-labels-v2",
    items: Array.from({length: 100}, (_, i) => ({id: `f-${i}`, kind: "experience", text, origin: "user", sourceRef, skillLabels: ["SQL"], requirementLevel: null})),
    selfAssessmentHints: Array.from({length: 100}, (_, i) => ({id: `h-${i}`, candidateId: `f-${i}`, skillLabel: "SQL", level: "guided"}))};
  expect(documentReviewDraftSchema.safeParse(draft).success).toBe(true);
  const durationsMs: number[] = [];
  for (let i = 0; i < 5; i++) {
    const start = performance.now();
    const kept = retain(draft.items, draft.selfAssessmentHints, draft.reviewProjection);
    durationsMs.push(performance.now() - start);
    expect(kept).toEqual(draft.selfAssessmentHints);
    expect(kept.every((hint, i) => hint === draft.selfAssessmentHints[i])).toBe(true);
  }
  const report = {items: 100, hints: 100, textLength: text.length, serializedBytes: Buffer.byteLength(JSON.stringify(draft)), durationsMs};
  writeFileSync("work/grounded-verification/edited-assessment-quality-20261108-bounded-cost-results.json", JSON.stringify(report, null, 2) + "\n");
  console.log(JSON.stringify({boundedCost: report}));
});
