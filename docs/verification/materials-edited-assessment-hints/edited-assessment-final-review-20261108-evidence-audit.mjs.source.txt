function auditRun(name, expectedExit) {
  const meta = json(folder + name + ".json"), log = read(folder + name + ".log");
  assert.equal(meta.name, name);
  assert.equal(meta.exit, expectedExit, name);
  assert.equal(meta.nonLoopbackDenials, 0, name);
  assert.equal(meta.logSha256, sha(log), name);
  const plain = stripVTControlCharacters(log.toString("utf8"));
  return { name, exit: meta.exit, nonLoopbackDenials: meta.nonLoopbackDenials, logSha256: meta.logSha256,
    testSummaries: [...plain.matchAll(/(?:Test Files|Tests)\s+[^\r\n]+|(?:#|ℹ) (?:tests|pass|fail|skipped) \d+/gu)].map(match => match[0]) };
}
const verifiedRuns = passes.map(name => auditRun(name, 0));
const failedRuns = preservedFailures.map(name => auditRun(name, 1));

const expectedSuitePaths = git("ls-files", "tests").split(/\r?\n/u).filter(path => /\.test\.(?:ts|tsx)$/u.test(path)).sort();
const suiteGroups = ["unit", "serial", "research"].map(group => {
  const name = `edited-assessment-full-${group}-20261108-b`;
  const plain = stripVTControlCharacters(read(folder + name + ".log").toString("utf8"));
  const paths = [...plain.matchAll(/^\s*✓\s+(tests\/[^\s]+\.test\.(?:ts|tsx))\s+\(/gmu)].map(match => match[1]);
  const summary = /Test Files\s+(\d+) passed \((\d+)\)[\s\S]*?Tests\s+(\d+) passed \((\d+)\)/u.exec(plain);
  assert.ok(summary, name);
  assert.equal(Number(summary[1]), Number(summary[2]), name);
  assert.equal(Number(summary[3]), Number(summary[4]), name);
  assert.equal(paths.length, Number(summary[1]), name);
  return { name, files: paths.length, passedTests: Number(summary[3]), paths };
});
