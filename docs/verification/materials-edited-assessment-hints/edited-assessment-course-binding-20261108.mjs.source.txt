import assert from "node:assert/strict";
import { readFile, writeFile } from "node:fs/promises";
import { join } from "node:path";
import { captureCourseReference, canonicalJson, sha256 } from "../../docs/verification/materials-quality-v2/capture-course-reference.mjs";
import { loadReviewedReferences } from "../../docs/verification/materials-quality-v2/candidate-references.mjs";

const root = process.cwd();
const priorPath = "docs/verification/materials-quality-v2/course-reference.snapshot.json";
const priorBytes = await readFile(join(root, priorPath));
assert.equal(sha256(priorBytes), "bf458308b5811f60b61f6e5fa24f6910642a7276615f0b212d930d04868d9676");
const prior = JSON.parse(priorBytes);
const current = await captureCourseReference({ root });
assert.deepEqual(current.objects, prior.objects, "Complete course exports must not change");
assert.deepEqual(current.inventory, prior.inventory);
assert.deepEqual(current.sourceFiles, prior.sourceFiles, "Full course source content must not change");
assert.deepEqual(current.dependencies.map(row => row.path), prior.dependencies.map(row => row.path));
const differences = current.dependencies.flatMap((row, index) => {
  const old = prior.dependencies[index];
  return canonicalJson(row) === canonicalJson(old) ? [] : [{ path: row.path, before: old, after: row }];
});
assert.equal(differences.length, 1);
assert.equal(differences[0].path, "app/contracts/planning.ts");
await assert.rejects(() => loadReviewedReferences(root), /reference-integrity: pinned bytes differ/u,
  "Original evaluator must still reject unreviewed source binding changes");
const snapshotName = "edited-assessment-course-capture-20261108-results.json";
const snapshotBytes = JSON.stringify(current, null, 2) + "\n";
await writeFile(join(root, "work/grounded-verification", snapshotName), snapshotBytes, { flag: "wx" });
const evidence = {
  status: "current-course-content-equivalent-new-dependency-binding-requires-review",
  priorReference: { path: priorPath, sha256: sha256(priorBytes) },
  currentCapture: { path: `work/grounded-verification/${snapshotName}`, sha256: sha256(snapshotBytes),
    note: "Raw output of the unchanged frozen capture utility; its historical id/date are not a new approved reference identity." },
  fullObjectsEqual: true, sourceFilesEqual: true, inventoryEqual: true,
  objectHashes: Object.fromEntries(Object.entries(current.objects).map(([name, value]) => [name, value.sha256])),
  dependencyCount: current.dependencies.length, unchangedDependencyCount: current.dependencies.length - differences.length,
  changedDependencies: differences,
  bundleEqual: current.capture.bundleSha256 === prior.capture.bundleSha256,
  legacyEvaluatorRejectsCurrentSource: true,
  newCandidateReferenceApproved: false, realModelQualificationPassed: false, gatePassed: false,
  modelCalls: 0, externalLinksFetched: 0,
};
const output = "work/grounded-verification/edited-assessment-course-binding-20261108-results.json";
await writeFile(join(root, output), JSON.stringify(evidence, null, 2) + "\n", { flag: "wx" });
console.log(JSON.stringify({ output, dependencyCount: evidence.dependencyCount, changedDependencies: differences.map(row => row.path),
  fullObjectsEqual: true, originalEvaluatorRejected: true, newCandidateReferenceApproved: false }));
