import { expect, it } from "vitest";
import { supportsMaterialSelfLevelHint } from "../../app/lib/materials/self-assessment-evidence";

it.each([
  ["按指导完成 SQL 查询练习", true],
  ["按照指导完成 SQL 查询练习", true],
  ["我按指导完成 SQL 查询练习", true],
  ["我按照指导完成 SQL 查询练习", true],
  ["按指导完成 ‘SQL’ 查询练习", true],
  ["按指导没有完成 SQL 查询练习", false],
  ["按指导未完成 SQL 查询练习", false],
  ["按指导计划完成 SQL 查询练习", false],
  ["按指导安排同事完成 SQL 查询练习", false],
  ["按指导由同事完成 SQL 查询练习", false],
  ["按指导同事完成 SQL 查询练习", false],
  ["按指导团队完成 SQL 查询练习", false],
  ["按指导将完成 SQL 查询练习", false],
  ["按指导记录 SQL 查询练习", false],
  ["示例：“按指导完成 SQL 查询练习”", false],
  ["没有按指导完成 SQL 查询练习", false],
  ["完成 SQL 查询练习", false],
] as const)("retains guidance only for supported personal action: %s", (text, supported) => {
  expect(supportsMaterialSelfLevelHint({ kind: "experience", text, labels: ["SQL"], level: "guided" })).toBe(supported);
});
