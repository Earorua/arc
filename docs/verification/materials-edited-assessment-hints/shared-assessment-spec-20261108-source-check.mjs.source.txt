const read = path => readFileSync(resolve(root, path), "utf8");
const printer = ts.createPrinter({ removeComments: true, newLine: ts.NewLineKind.LineFeed });
function parse(path) {
  const file = ts.createSourceFile(path, read(path), ts.ScriptTarget.Latest, true, ts.ScriptKind.TS);
  const declarations = new Map();
  for (const statement of file.statements) {
    if ((ts.isFunctionDeclaration(statement) || ts.isTypeAliasDeclaration(statement)) && statement.name) {
      declarations.set(statement.name.text, statement);
    } else if (ts.isVariableStatement(statement)) {
      assert.equal(statement.declarationList.declarations.length, 1, "probe expects one named declaration per statement");
      const declaration = statement.declarationList.declarations[0];
      assert(ts.isIdentifier(declaration.name));
      declarations.set(declaration.name.text, statement);
    }
  }
  return { file, declarations };
}

let negationGuard;
function canonical(container, name, removeGuard = false) {
  let node = container.declarations.get(name);
  assert(node, `missing declaration ${name}`);
  if (removeGuard) {
    assert(ts.isFunctionDeclaration(node) && node.body);
    const guards = node.body.statements.filter(s => ts.isIfStatement(s) && s.getText(container.file).includes("did\\s+not"));
    assert.equal(guards.length, 1, "exactly one new actor-local did-not guard");
    negationGuard = guards[0].getText(container.file);
    node = ts.factory.updateFunctionDeclaration(node, node.modifiers, node.asteriskToken, node.name,
      node.typeParameters, node.parameters, node.type,
      ts.factory.updateBlock(node.body, node.body.statements.filter(s => s !== guards[0])));
  }
  let output = printer.printNode(ts.EmitHint.Unspecified, node, container.file).replace(/^export /u, "");
  for (const [oldName, newName] of renamed) output = output.replace(new RegExp(`\\b${newName}\\b`, "gu"), oldName);
  output = output.replace("hasPersonalPracticeNeed(quote, masked)", "assessmentScopes(quote, masked).some(({ text }) => personalPracticeNeed(text))");
  return output;
}

const edges = [], files = new Map(), external = new Set();
function visit(absolute, ancestors = []) {
  const local = relative(root, absolute).replaceAll("\\", "/");
  assert(!ancestors.includes(local), `runtime import cycle: ${[...ancestors, local].join(" -> ")}`);
  if (files.has(local)) return;
  assert(!/(?:^|\/)(?:worker|server)(?:\/|$)/u.test(local), `forbidden runtime dependency ${local}`);
  const text = readFileSync(absolute, "utf8");
  assert(!/\bprocess\.env\b|\bfetch\s*\(|\bD1Database\b/u.test(text), `environment/network/D1 coupling in ${local}`);
  const file = ts.createSourceFile(local, text, ts.ScriptTarget.Latest, true, ts.ScriptKind.TS);
  files.set(local, createHash("sha256").update(text).digest("hex"));
  for (const statement of file.statements) {
    if (!ts.isImportDeclaration(statement) || statement.importClause?.isTypeOnly) continue;
    const specifier = statement.moduleSpecifier.text;
    if (!specifier.startsWith(".")) {
      external.add(specifier);
      continue;
    }
    const base = resolve(dirname(absolute), specifier);
    const target = [base, base + ".ts", resolve(base, "index.ts")].find(candidate => existsSync(candidate));
    assert(target, `unresolved import ${specifier} from ${local}`);
    edges.push([local, relative(root, target).replaceAll("\\", "/")]);
    visit(target, [...ancestors, local]);
  }
}
