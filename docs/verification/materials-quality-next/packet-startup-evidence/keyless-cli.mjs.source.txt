function fixtureFile(name, object) {
  const bytes = JSON.stringify(object, null, 2) + '\n';
  writeFileSync(join(directory, name), bytes, { flag: 'wx' });
  return { path: name, sha256: sha(bytes) };
}

function inspectLedger() {
  const db = new DatabaseSync(ledgerFile, { readOnly: true });
  try { return { identity: sha(JSON.stringify([ledgerFile, db.prepare('SELECT id FROM ledger_identity WHERE singleton=1').get().id])),
    runs: db.prepare('SELECT count(*) AS n FROM collection_runs').get().n, attempts: db.prepare('SELECT count(*) AS n FROM attempts').get().n }; }
  finally { db.close(); }
}
