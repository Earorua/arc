import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { readFileSync, writeFileSync } from 'node:fs';
import { stripVTControlCharacters } from 'node:util';
const dir = 'work/grounded-verification/', prefix = 'course-relations-final-20261119-';
const json = path => JSON.parse(readFileSync(path, 'utf8'));
const hash = path => createHash('sha256').update(readFileSync(path)).digest('hex');
const source = json(dir + prefix + 'source.json');
for (const [path, digest] of Object.entries(source.files)) assert.equal(hash(path), digest, path);
const runs = Object.fromEntries(['vitest-01', 'node-01', 'types-01', 'lint-01', 'build-01', 'rendered-01'].map(name => {
  const path = dir + prefix + name, receipt = json(path + '.json');
  assert.equal(receipt.exit, 0, name); assert.equal(receipt.nonLoopbackDenials, 0, name);
  assert.equal(hash(path + '.log'), receipt.logSha256, name);
  return [name, receipt];
}));
const plain = name => stripVTControlCharacters(readFileSync(dir + prefix + name + '.log', 'utf8'));
const vitestText = plain('vitest-01');
const rows = [...vitestText.matchAll(/^\s*[✓✔]\s+(tests\/\S+\.test\.(?:ts|tsx))\s+\((\d+) tests?\)/gm)]
  .map(([, file, count]) => ({ file, tests: Number(count) }));
assert.deepEqual(rows.map(row => row.file).sort(), source.vitestFiles);
const suiteSummary = /Test Files\s+(\d+) passed \((\d+)\)/u.exec(vitestText);
const testSummary = /Tests\s+(\d+) passed \((\d+)\)/u.exec(vitestText);
assert.ok(suiteSummary && testSummary, 'All-pass summary without skipped or failed cases');
assert.equal(Number(suiteSummary[1]), rows.length); assert.equal(suiteSummary[1], suiteSummary[2]);
const vitestCount = rows.reduce((sum, row) => sum + row.tests, 0);
assert.equal(Number(testSummary[1]), vitestCount); assert.equal(testSummary[1], testSummary[2]);
function nodeCounts(name) {
  const text = plain(name), result = {};
  for (const key of ['tests', 'pass', 'fail', 'cancelled', 'skipped', 'todo']) {
    const match = new RegExp('^[ℹ#] ' + key + ' (\\d+)', 'm').exec(text);
    assert.ok(match, name + ': ' + key); result[key] = Number(match[1]);
  }
  assert.equal(result.tests, result.pass);
  for (const key of ['fail', 'cancelled', 'skipped', 'todo']) assert.equal(result[key], 0);
  return result;
}
const node = nodeCounts('node-01'), rendered = nodeCounts('rendered-01');
assert.equal(rendered.tests, 5);
